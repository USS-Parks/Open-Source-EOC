type TaskStatus = "completed" | "unavailable";

  status: TaskStatus;
  route: string;
  permission: string;
  syntheticData: Record<string, unknown>;
  startCondition: string;

  elapsedMs: number | null;
  interactions: number | null;
  discoveryInteractions?: number;
  duplicateEntryCount: number | null;
  assistanceCount: number | null;
  errorCount: number;
  limitation: string | null;
}

function chromiumPath(): string {
  const candidates = [
    process.env.OPENEOC_CHROMIUM,
    "/opt/pw-browsers/chromium",
    "/usr/bin/google-chrome",
    "/usr/bin/chromium-browser",
    "/usr/bin/chromium",
  ];
  for (const candidate of candidates) if (candidate && existsSync(candidate)) return candidate;
  throw new Error("no Chromium found; set OPENEOC_CHROMIUM");
}

async function login(email: string, password: string): Promise<string> {
  const response = await app.inject({
    method: "POST",
    url: "/api/v1/auth/login",
    payload: { email, password },
  });
  if (response.statusCode !== 200) throw new Error(`login failed: ${response.body}`);
  return response.json().accessToken as string;
}

async function post(url: string, payload: Record<string, unknown>): Promise<Record<string, unknown>> {
  const response = await app.inject({
    method: "POST",
    url,
    headers: { authorization: `Bearer ${adminToken}` },
    payload,
  });
  if (response.statusCode < 200 || response.statusCode >= 300) {
    throw new Error(`POST ${url} failed: ${response.statusCode} ${response.body}`);
  }
  return response.json() as Record<string, unknown>;
}

async function createBoard(templateKey: string): Promise<string> {
  const result = await post(`/api/v1/jurisdictions/${jurisdictionId}/boards`, { templateKey });
  return result.id as string;
}

async function addRecord(boardId: string, data: Record<string, unknown>): Promise<void> {
  await post(`/api/v1/boards/${boardId}/records`, data);
}

async function completed(
  metric: Omit<TaskMetric, "status" | "elapsedMs" | "interactions" | "errorCount" | "limitation">,
  body: (act: (operation: () => Promise<unknown>) => Promise<void>) => Promise<void>,
): Promise<void> {
  const errorsBefore = pageErrors.length;
  let interactions = 0;
  const start = performance.now();
  await body(async (operation) => {
    interactions += 1;
    await operation();
  });
  report.tasks.push({
    ...metric,
    status: "completed",
    elapsedMs: Math.round((performance.now() - start) * 10) / 10,
    interactions,
    errorCount: pageErrors.length - errorsBefore,
    limitation: "Headless scripted timing is environment-specific and is not a human operator time.",
  });
}

function unavailable(
  metric: Omit<TaskMetric, "status" | "elapsedMs" | "interactions" | "errorCount" | "limitation">,
  limitation: string,
): void {
  report.tasks.push({
    ...metric,
    status: "unavailable",
    elapsedMs: null,
    interactions: null,
    errorCount: 0,
    limitation,
  });
}

beforeAll(async () => {
  const webDir = join(process.cwd(), "web");
  const publicDir = join(webDir, "public");
  const { build } = await import("../../../../web/node_modules/vite/dist/node/index.js");
  await build({
    root: webDir,
    base: "./",
    publicDir: false,
    logLevel: "silent",
    build: { outDir: DIST, emptyOutDir: true },
  });
  expect(existsSync(join(DIST, "index.html"))).toBe(true);

  ({ admin, runtime } = await freshDb());
  const seed = await seedIdentity(admin);
  jurisdictionId = seed.jurisdictionId;
  await ensureStandardTemplates(admin);
  await ensureStandardIncidentTemplates(admin);
  app = buildApp(runtime, { oidc: null });

  app.get("/app/*", (request, reply) => {
    const relative = (request.params as { "*": string })["*"] || "index.html";
    const safe = relative.replaceAll("..", "");
    let path = join(DIST, safe);
    if (!existsSync(path)) path = join(publicDir, safe);
    if (!existsSync(path)) return reply.status(404).send("missing");
    const extension = path.slice(path.lastIndexOf("."));
    const buffer = readFileSync(path);
    const range = request.headers.range;
    const match = range ? /^bytes=(\d+)-(\d*)$/.exec(range) : null;
    if (match) {
      const start = Number(match[1]);
      const end = match[2] ? Number(match[2]) : buffer.length - 1;
      const slice = buffer.subarray(start, Math.min(end, buffer.length - 1) + 1);
      return reply
        .status(206)
        .header("content-type", TYPES[extension] ?? "application/octet-stream")
        .header("accept-ranges", "bytes")
        .header("content-range", `bytes ${start}-${start + slice.length - 1}/${buffer.length}`)
        .send(slice);
    }
    return reply
      .header("content-type", TYPES[extension] ?? "application/octet-stream")
      .header("accept-ranges", "bytes")
      .send(buffer);
  });
  await app.listen({ port: 0, host: "127.0.0.1" });
  const address = app.server.address();
  baseUrl = `http://127.0.0.1:${typeof address === "object" && address ? address.port : 0}`;
  adminToken = await login("admin@example.org", "correct-horse-battery");

  const fieldReports = await createBoard("field_reports");
  await createBoard("lifelines");
  await createBoard("significant_events");
  await createBoard("resource_request");
  const activation = await post(`/api/v1/jurisdictions/${jurisdictionId}/incidents`, {
    templateKey: "wildfire",
    name: "D01 North Coast Exercise",
  });
  const incidentId = activation.incidentId as string;
  await admin`insert into incident_boards (incident_id, board_id) values (${incidentId}, ${fieldReports})`;
  await addRecord(fieldReports, {
    summary: "Bridge approach passable for emergency vehicles",
    category: "hazard",
    location: { type: "Point", coordinates: [-123.61, 41.29] },
  });
  const lifeline = await app.inject({
    method: "PUT",
    url: `/api/v1/jurisdictions/${jurisdictionId}/lifelines`,
    headers: { authorization: `Bearer ${adminToken}` },
    payload: { lifeline: "energy", status: "unstable", note: "Two substations offline; service disrupted" },
  });
  expect(lifeline.statusCode).toBe(200);
  await post(`/api/v1/jurisdictions/${jurisdictionId}/sitreps`, {
    period: "OP-D01",
    incidentId,
  });

  browser = await chromium.launch({ executablePath: chromiumPath(), args: ["--no-sandbox"] });
  page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
  page.on("pageerror", (error) => pageErrors.push(error.message));
  await page.route("**/*", (route) => {
    const url = route.request().url();
    if (url.startsWith(baseUrl) || url.startsWith("data:") || url.startsWith("blob:")) return route.continue();
    externalRequests.push(url);
    return route.abort();
  });
  await page.goto(`${baseUrl}/app/index.html`, { waitUntil: "load" });
  await page.getByLabel("Email").fill("admin@example.org");
  await page.getByLabel("Password").fill("correct-horse-battery");
  await page.getByRole("button", { name: "Sign in" }).click();
  await page.waitForSelector('[data-testid="cop-map"]', { timeout: 20_000 });
}, 120_000);

afterAll(async () => {
  report.measuredAt = new Date().toISOString();
  report.runErrors.push(...pageErrors.map((message) => `browser: ${message}`));
  report.runErrors.push(...externalRequests.map((url) => `external request blocked: ${url}`));
  mkdirSync(dirname(OUTPUT), { recursive: true });
  writeFileSync(OUTPUT, `${JSON.stringify(report, null, 2)}\n`, "utf8");
  await page?.close();
  await browser?.close();
  await app?.close();
  await runtime?.end();
  await admin?.end();
});
