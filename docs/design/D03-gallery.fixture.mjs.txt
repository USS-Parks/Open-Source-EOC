import { existsSync, mkdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const { chromium } = await import("../../../../server/node_modules/playwright-core/index.mjs");

const reviewUrl = process.env.D03_REVIEW_URL ?? "http://127.0.0.1:5175/docs/design/D03-gallery.html";
const reviewOrigin = new URL(reviewUrl).origin;
const reviewPhase = process.env.D03_REVIEW_PHASE ?? "full";
const shotDir = process.env.D03_SHOT_DIR ?? "C:/Users/17076/Documents/Open Source EOC/deploy/test-runtime/out/d03-review/shots";
const resultPath = process.env.D03_RESULT ?? "C:/Users/17076/Documents/Open Source EOC/deploy/test-runtime/out/d03-review/result.json";
const chromiumCandidates = [
  process.env.OPENEOC_CHROMIUM,
  "C:/Program Files/Google/Chrome/Application/chrome.exe",
  "C:/Program Files (x86)/Google/Chrome/Application/chrome.exe",
].filter(Boolean);
const executablePath = chromiumCandidates.find((candidate) => existsSync(candidate));
if (!executablePath) throw new Error("No Chromium found; set OPENEOC_CHROMIUM");

mkdirSync(shotDir, { recursive: true });
const browser = await chromium.launch({ executablePath, headless: true });
const context = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 });
const page = await context.newPage();
const pageErrors = [];
const externalRequests = [];
page.on("pageerror", (error) => pageErrors.push(String(error)));
page.on("request", (request) => {
  const url = request.url();
  if (!url.startsWith(`${reviewOrigin}/`)) externalRequests.push(url);
});
await context.route("**/*", async (route) => {
  const url = route.request().url();
  if (url.startsWith(`${reviewOrigin}/`)) await route.continue();
  else await route.abort();
});

function assert(condition, message) {
  if (!condition) throw new Error(message);
}

async function setTheme(theme) {
  const current = await page.locator(".composition-review").getAttribute("data-review-theme");
  if (current !== theme) await page.getByRole("button", { name: theme === "dark" ? "Use dark theme" : "Use light theme" }).click();
  assert(await page.locator(".composition-review").getAttribute("data-review-theme") === theme, `Theme did not change to ${theme}`);
}

async function setScreen(label) {
  await page.getByLabel("Screen").selectOption({ label });
  await page.locator(".composition-page-header h1:visible").waitFor();
}

async function capture(name) {
  await page.screenshot({ path: join(shotDir, `${name}.png`), fullPage: false });
}

async function assertNoPageOverflow(label) {
  const widths = await page.evaluate(() => {
    const shell = document.querySelector(".composition-shell");
    return {
      viewport: window.innerWidth,
      document: document.documentElement.scrollWidth,
      shellClient: shell?.clientWidth ?? 0,
      shellScroll: shell?.scrollWidth ?? 0,
    };
  });
  assert(widths.document <= widths.viewport + 1, `${label} has page overflow: ${widths.document}px > ${widths.viewport}px`);
  assert(widths.shellScroll <= widths.shellClient + 1, `${label} overflows the operator shell: ${widths.shellScroll}px > ${widths.shellClient}px`);
}

async function assertLifelineContainment() {
  const geometry = await page.evaluate(() => {
    const workspace = document.querySelector(".composition-workspace");
    const badges = Array.from(document.querySelectorAll(".composition-lifeline-card > .composition-condition")).map((badge) => {
      const bounds = badge.getBoundingClientRect();
      return { text: badge.textContent?.trim() ?? "", left: bounds.left, right: bounds.right };
    });
    return { workspaceClient: workspace?.clientWidth ?? 0, workspaceScroll: workspace?.scrollWidth ?? 0, viewport: window.innerWidth, badges };
  });
  assert(geometry.workspaceScroll <= geometry.workspaceClient + 1, `Narrow Lifelines overflows its workspace: ${geometry.workspaceScroll}px > ${geometry.workspaceClient}px`);
  const clipped = geometry.badges.filter((badge) => badge.left < 0 || badge.right > geometry.viewport + 1);
  assert(clipped.length === 0, `Lifeline conditions leave the viewport: ${clipped.map((badge) => badge.text).join(", ")}`);
}

async function activeControlName() {
  return page.evaluate(() => {
    const control = document.activeElement;
    if (!(control instanceof HTMLElement)) return "";
    const labelledBy = control.getAttribute("aria-labelledby");
    const labelledText = labelledBy ? document.getElementById(labelledBy)?.textContent : "";
    const labels = "labels" in control ? Array.from(control.labels ?? []).map((label) => label.textContent).join(" ") : "";
    return (control.getAttribute("aria-label") || labelledText || labels || control.textContent || "").trim();
  });
}

async function tabToControl(name) {
  for (let index = 0; index < 80; index += 1) {
    await page.keyboard.press("Tab");
    if ((await activeControlName()).includes(name)) return;
  }
  throw new Error(`Could not reach ${name} with the keyboard`);
}
