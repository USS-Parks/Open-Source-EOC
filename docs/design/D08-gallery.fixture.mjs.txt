import { existsSync, mkdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const { chromium } = await import("../../../../server/node_modules/playwright-core/index.mjs");
const reviewUrl = process.env.D08_REVIEW_URL ?? "http://127.0.0.1:5175/docs/design/D08-gallery.html";
const reviewOrigin = new URL(reviewUrl).origin;
const shotDir = process.env.D08_SHOT_DIR ?? "C:/Users/17076/Documents/Open Source EOC/deploy/test-runtime/out/d08-review/shots";
const resultPath = process.env.D08_RESULT ?? "C:/Users/17076/Documents/Open Source EOC/deploy/test-runtime/out/d08-review/result.json";
const executablePath = [process.env.OPENEOC_CHROMIUM, "C:/Program Files/Google/Chrome/Application/chrome.exe", "C:/Program Files (x86)/Google/Chrome/Application/chrome.exe"].filter(Boolean).find((candidate) => existsSync(candidate));
if (!executablePath) throw new Error("No Chromium found; set OPENEOC_CHROMIUM");

mkdirSync(shotDir, { recursive: true });
const browser = await chromium.launch({ executablePath, headless: true });
const context = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 });
const page = await context.newPage();
const pageErrors = [];
const externalRequests = [];
page.on("pageerror", (error) => pageErrors.push(String(error)));
page.on("request", (request) => { if (!request.url().startsWith(`${reviewOrigin}/`)) externalRequests.push(request.url()); });
await context.route("**/*", async (route) => {
  if (route.request().url().startsWith(`${reviewOrigin}/`)) await route.continue();
  else await route.abort();
});

function assert(condition, message) { if (!condition) throw new Error(message); }
async function capture(name) { await page.screenshot({ path: join(shotDir, `${name}.png`), fullPage: true }); }
async function assertContained(label) {
  const geometry = await page.evaluate(() => ({ viewport: window.innerWidth, documentWidth: document.documentElement.scrollWidth, drawerWidth: document.querySelector(".eoc-drawer")?.getBoundingClientRect().width ?? 0, drawerClient: document.querySelector(".eoc-drawer")?.clientWidth ?? 0, drawerScroll: document.querySelector(".eoc-drawer")?.scrollWidth ?? 0 }));
  assert(geometry.documentWidth <= geometry.viewport + 1, `${label} page overflow: ${geometry.documentWidth}px > ${geometry.viewport}px`);
  if (geometry.drawerClient) assert(geometry.drawerScroll <= geometry.drawerClient + 1, `${label} drawer overflow: ${geometry.drawerScroll}px > ${geometry.drawerClient}px`);
  if (geometry.viewport > 640 && geometry.drawerWidth) assert(geometry.drawerWidth === 340, `${label} drawer width drifted from approved 340px: ${geometry.drawerWidth}px`);
}
async function setTheme(theme) {
  const current = await page.locator(".form-review").getAttribute("data-review-theme");
  if (current !== theme) await page.getByRole("button", { name: theme === "dark" ? "Use dark theme" : "Use light theme" }).click();
  assert(await page.locator(".form-review").getAttribute("data-review-theme") === theme, `Theme did not change to ${theme}`);
}
async function openDrawer() {
  await page.getByRole("button", { name: "New request" }).click();
  await page.getByRole("dialog", { name: "New resource request" }).waitFor();
  await page.getByText("Loading saved draft…").waitFor({ state: "detached" });
}
async function closeKeepDraft() {
  await page.getByRole("button", { name: "Close and keep draft" }).click();
  await page.getByRole("dialog", { name: "New resource request" }).waitFor({ state: "detached" });
}

try {
  await page.goto(reviewUrl, { waitUntil: "networkidle" });
  await page.locator(".form-review").waitFor();
  await openDrawer();
  await page.getByRole("button", { name: "Submit request" }).click();
  const summary = page.getByText(/Correct \d+ form errors/);
  await summary.waitFor();
  assert(await summary.evaluate((element) => element.parentElement === document.activeElement), "Validation summary did not receive focus");
  const summaryLink = page.getByRole("link", { name: /Request summary:/ });
  assert(await summaryLink.isVisible(), "Validation summary is not linked to Request summary");
  const hashBeforeCorrection = await page.evaluate(() => window.location.hash);
  await summaryLink.click();
  assert(await page.evaluate(() => window.location.hash) === hashBeforeCorrection, "Validation correction mutated the application hash route");
  assert(await page.getByLabel("Request summary").evaluate((element) => element === document.activeElement), "Validation correction did not focus Request summary");
  await assertContained("Wide light validation");
  await capture("wide-light-validation");

  await page.getByLabel("Request summary").fill("Generator support for Wendy’s Shelter");
  await page.getByLabel("Quantity").fill("4");
  await page.getByLabel("Urgent request").check();
  assert(await page.getByLabel("Priority").isVisible(), "Conditional Priority did not appear");
  await page.getByLabel("Priority").selectOption("urgent");
  await page.getByLabel("Requester").selectOption({ label: "Jordan Diaz · Planning" });
  assert((await page.getByLabel("Request summary").inputValue()).includes("Generator support"), "Correcting errors lost the completed summary");

  const closeButton = page.getByRole("button", { name: "Close New resource request" });
  await closeButton.click();
  const guard = page.getByRole("alertdialog", { name: "Discard unsaved changes?" });
  await guard.waitFor();
  assert(await page.evaluate(() => document.activeElement?.textContent?.trim()) === "Keep editing", "Unsaved guard did not focus Keep editing");
  await page.keyboard.press("Escape");
  assert(await guard.count() === 0, "Escape did not return to editing");
  assert(await closeButton.evaluate((element) => element === document.activeElement), "Unsaved guard did not restore focus to its invoking control");

  const footerClose = page.getByRole("button", { name: "Close and keep draft" });
  await footerClose.focus();
  await page.keyboard.press("Tab");
  assert((await page.evaluate(() => document.activeElement?.getAttribute("aria-label"))) === "Close New resource request", "Drawer focus did not wrap to its first control");
  await closeKeepDraft();
  await openDrawer();
  assert((await page.getByLabel("Request summary").inputValue()) === "Generator support for Wendy’s Shelter", "Draft resume required summary re-entry");
  assert((await page.getByLabel("Quantity").inputValue()) === "4", "Draft resume required quantity re-entry");

  await closeKeepDraft();
  await setTheme("dark");
  await openDrawer();
  await assertContained("Wide dark resumed draft");
  await capture("wide-dark-resumed-draft");
  await page.getByRole("button", { name: "Submit request" }).click();
  await page.getByText("Record was not saved.").waitFor();
  assert((await page.getByLabel("Request summary").inputValue()).includes("Generator support"), "Rejected submit cleared form values");
  assert(await page.getByText("Synthetic service rejection").isVisible(), "Rejected submit reason is missing");
  assert(await page.getByText("Record saved").count() === 0, "Rejected submit displayed false success");

  await page.setViewportSize({ width: 390, height: 844 });
  await assertContained("Narrow dark rejected submit");
  await capture("narrow-dark-rejected-submit");
  await closeKeepDraft();
  await page.getByLabel("Reject synthetic submit").uncheck();
  await openDrawer();
  await page.getByRole("button", { name: "Submit request" }).click();
  await page.getByText("Record saved").waitFor();
  assert((await page.getByLabel("Request summary").inputValue()).includes("Generator support"), "Successful submit changed entered values");
  await page.getByRole("button", { name: "Close New resource request" }).click();
  await page.getByRole("dialog", { name: "New resource request" }).waitFor({ state: "detached" });

  await setTheme("light");
  const modalOpener = page.getByRole("button", { name: "Open confirmation dialog" });
  await modalOpener.focus();
  await modalOpener.click();
  const modal = page.getByRole("dialog", { name: "Publish operational update" });
  await modal.waitFor();
  await page.keyboard.press("Escape");
  await page.getByRole("alertdialog", { name: "Discard unsaved changes?" }).waitFor();
  await assertContained("Narrow light unsaved guard");
  await capture("narrow-light-unsaved-guard");
  await page.getByRole("button", { name: "Discard changes" }).click();
  await modal.waitFor({ state: "detached" });
  assert(await modalOpener.evaluate((element) => element === document.activeElement), "Modal did not restore focus to its opener");

  assert(pageErrors.length === 0, `Page errors: ${pageErrors.join(" | ")}`);
  assert(externalRequests.length === 0, `External requests: ${externalRequests.join(" | ")}`);
  const result = {
    status: "passed",
    reviewUrl,
    screenshots: 4,
    viewports: ["1440x900", "390x844"],
    themes: ["light", "dark"],
    validationCorrectionChecked: true,
    conditionalControlChecked: true,
    draftResumeWithoutReentryChecked: true,
    rejectedSubmitPreservedChecked: true,
    successfulSubmitTruthChecked: true,
    focusTrapRestoreChecked: true,
    unsavedGuardChecked: true,
    horizontalOverflowChecked: true,
    pageErrors,
    externalRequests,
  };
  writeFileSync(resultPath, JSON.stringify(result, null, 2));
  console.log(JSON.stringify(result));
} catch (error) {
  console.error(JSON.stringify({ status: "failed", error: String(error), pageErrors, externalRequests }));
  throw error;
} finally {
  await browser.close();
}
